30ParserResult AbstractNodeParser::parsePackage(const QSharedPointer<PKG::AbstractData> &pkg,
31 const Header &pkgHeader,
32 AbstractNodeInfo *sender) {
33 auto nodePtr = node();
34 if (!nodePtr) {
35 return ParserResult::NotProcessed;
36 }
37
38 if (!(sender)) {
39 qCritical() << "sender socket is not valid!";
40 return ParserResult::Error;
41 }
42
43 if (!pkg->isValid()) {
44 qCritical() << "incomming package is not valid!";
45
46 nodePtr->changeTrust(sender->networkAddress(), CRITICAL_ERROOR);
47 return ParserResult::Error;
48 }
49
50 if (PKG::Ping::command() == pkg->cmd()) {
51 auto cmd = pkg.staticCast<PKG::Ping>();
52 if (!cmd->ansver()) {
53 cmd->setAnsver(true);
54 nodePtr->sendData(cmd.data(), sender, &pkgHeader);
55 }
56
57 emit sigPingReceived(cmd);
58
59 return ParserResult::Processed;
60 } else if (PKG::BadRequest::command() == pkg->cmd()) {
61 auto cmd = static_cast<PKG::BadRequest *>(pkg.data());
62
63 emit nodePtr->requestError(cmd->errCode(), cmd->err());
64
65 return ParserResult::Processed;
66
67 } else if (PKG::CloseConnection::command() == pkg->cmd()) {
68 if (sender->isLocal()) {
69 nodePtr->removeNode(sender->networkAddress());
70 }
71 return ParserResult::Processed;
72 }
73
74 return ParserResult::NotProcessed;
75}
