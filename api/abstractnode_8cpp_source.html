136void AbstractNode::stop() {
137 close();
138
139 _connectionsMutex.lock();
140 for (const auto &i : std::as_const(_connections)) {
141 i->disconnect();
142 }
143 _connectionsMutex.unlock();
144
145 // Pool workers can be blocked while they wait for a job that was posted
146 // to the main thread (for example, socket writes executed by the data
147 // sender). Only this thread can deliver those jobs, so it must keep
148 // processing events until all workers are finished. Otherwise this
149 // method would wait for the workers and the workers would wait for this
150 // thread (deadlock).
151 const auto deadline = QDeadlineTimer(WAIT_TIME);
152 while (!deadline.hasExpired()) {
153 bool hasWork = false;
154 _workersMutex.lock();
155 for (const auto &worker : _workers) {
156 if (worker && !worker->isFinished()) {
157 hasWork = true;
158 break;
159 }
160 }
161 _workersMutex.unlock();
162
163 if (!hasWork) {
164 break;
165 }
166
167 QCoreApplication::processEvents(QEventLoop::AllEvents, 100);
168 }
169
170 _workersMutex.lock();
171 for (auto it: std::as_const(_workers)) {
172 if (!it->isFinished()) {
173 it->cancel();
174 it->waitForFinished();
175 }
176 }
177 _workersMutex.unlock();
178
179 deinitThreadPool();
180}

290 } else {
291 addNode(HostAddress{addresses.first(), port});

503bool AbstractNode::registerSocket(QAbstractSocket *socket, const HostAddress* clientAddress) {
504
505 if (connectionsCount() >= maxPendingConnections()) {
506 return false;
507 }
508
509 HostAddress cliAddress;
510 if (clientAddress)
511 cliAddress = *clientAddress;
512 else
513 cliAddress = HostAddress{socket->peerAddress(), socket->peerPort()};
514
515
516 _connectionsMutex.lock();
517
518 if (_connections.contains(cliAddress)) {
519 auto info =_connections.value(cliAddress);
520 info->setSct(socket);
521 info->setIsLocal(clientAddress);
522
523 _connectionsMutex.unlock();
524
525 if (!info->isValid()) {
526 return false;
527 }
528
529 nodeAddedSucessful(info);
530 return true;
531 }
532
533 auto info = createNodeInfo(socket, &cliAddress);
534
535 info->setIsLocal(clientAddress);
536
537 _connections[cliAddress] = info;
538
539 _connectionsMutex.unlock();
540
541 connect(info, &AbstractNodeInfo::sigReadyRead,
542 this, &AbstractNode::avelableBytes, Qt::DirectConnection);
543
544 // using direct connection because socket clear all data of ip and port after disconnected.
545 connect(info, &AbstractNodeInfo::statusChaned,
546 this, &AbstractNode::handleNodeStatusChanged,
547 Qt::QueuedConnection);
548
549 if (info->status() != NodeCoonectionStatus::NotConnected) {
550 handleNodeStatusChanged(info, info->status());
551 }
552
553 // using direct connection because socket clear all data of ip and port after disconnected.
554 connect(info, &AbstractNodeInfo::sigErrorOccurred,
555 this, &AbstractNode::nodeErrorOccured,
556 Qt::QueuedConnection);
557
558 // check node confirmed
559 QTimer::singleShot(WAIT_TIME, this, [this, cliAddress]() {
560 checkConfirmendOfNode(getInfoPtr(cliAddress));
561 });
562
563 nodeAddedSucessful(info);
564
565 return true;
566}

607unsigned int AbstractNode::sendData(const PKG::AbstractData *resp,
608 const AbstractNodeInfo *node,
609 const Header *req) {
610
611 if (!node) {
612 qDebug() << "Response not sent because client == null";
613 return 0;
614 }
615
616 if (!resp) {
617 return 0;
618 }
619
620 Package pkg;
621 bool convert = false;
622 if (req && req->isValid()) {
623 convert = resp->toPackage(pkg, node->multiVersionPackages().value(resp->cmd()),req->hash);
624 } else {
625 convert = resp->toPackage(pkg, node->multiVersionPackages().value(resp->cmd()));
626 }
627
628 if (!convert) {
629
630 if (static_cast<unsigned int>(pkg.data.size()) > Package::maximumSize()) {
631 // big data
632
633 auto wrap = QSharedPointer<BigDataWraper>::create();
634 wrap->setData(resp);
635 if ( parsePackage(wrap, {}, getInfoPtr(node->networkAddress())) != ParserResult::Processed) {
636 return 0;
637 }
638
639 return BIG_DATA_HASH_ID;
640 }
641 qCritical() << "Response not sent because dont create package from object";
642 return 0;
643 }
644
645 if (!sendPackage(pkg, node->sct())) {
646 qCritical() << "Response not sent!";
647 return 0;
648 }
649
650 return pkg.hdr.hash;
651}

828void AbstractNode::avelableBytes(AbstractNodeInfo *sender) {
829
830 if (!sender) {
831 return;
832 }
833
834 auto id = sender->networkAddress();
835
836 if (!_connections.contains(id)) {
837 return;
838 }
839
840 if (!_receiveData.contains(id)) {
841 _receiveData.insert(id, new ReceiveData());
842 }
843
844 auto &pkg = _receiveData[id]->_pkg;
845 auto &hdrArray = _receiveData[id]->_hdrArray;
846
847 int workIndex = 0;
848 const int headerSize = sizeof(Header);
849
850 auto socket = sender->sct();
851 if (!socket) {
852 pkg.reset();
853 hdrArray.clear();
854 return;
855 }
856
857 // concat with old data of header.
858 auto array = hdrArray;
859 while (socket->bytesAvailable() > 0) {
860 array += socket->readAll();
861 };
862
863 const int arraySize = array.size();
864 hdrArray.clear();
865
866 while (arraySize > workIndex) {
867
868 int offset = arraySize - workIndex;
869
870 if (pkg.hdr.isValid()) {
871 // CASE 1: The Package data is still not collected, but the header is already collected. performs full or partial filling of packet data.
872
873 int dataLength = std::min(static_cast<int>(pkg.hdr.size - pkg.data.size()),
874 arraySize - workIndex);
875 pkg.data.append(array.mid(workIndex, dataLength));
876
877 workIndex += dataLength;
878
879
880 } else if (offset >= headerSize) {
881
882 // CASE 2: The header and package still do not exist and the amount of data allows you to create a new header. A header is created and will fill in all or part of the package data.
883
884 pkg.reset();
885
886 memcpy(&pkg.hdr,
887 array.data() + workIndex, headerSize);
888
889 if (!pkg.hdr.isValid())
890 return;
891
892 int dataLength = std::min(static_cast<int>(pkg.hdr.size),
893 arraySize - headerSize - workIndex);
894
895 pkg.data.append(array.mid(workIndex + headerSize, dataLength));

899 } else {

901
902 unsigned char dataLength = static_cast<unsigned char>(arraySize - workIndex);
903 hdrArray += array.mid(workIndex, dataLength);
904 workIndex += dataLength;
905 }
906
907 if (pkg.isValid()) {
908 newWork(pkg, sender, id);
909 pkg.reset();
910 hdrArray.clear();
911 } else if (static_cast<unsigned int>(pkg.data.size()) >= pkg.hdr.size) {
912 qWarning() << "Invalid Package received." + pkg.toString();
913 pkg.reset();
914 hdrArray.clear();
915 changeTrust(id, CRITICAL_ERROOR);
916
917 }
918 }

1102 }
